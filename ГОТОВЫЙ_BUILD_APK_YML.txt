name: Build SUPER NIKOLIO Android TV APK

on:
  push:
    branches: [main]
  workflow_dispatch:

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout project
        uses: actions/checkout@v4
      - name: Setup JDK 17
        uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '17'
      - name: Install Android SDK packages
        shell: bash
        run: |
          SDKMANAGER="${ANDROID_HOME:-$ANDROID_SDK_ROOT}/cmdline-tools/latest/bin/sdkmanager"
          if [ ! -x "$SDKMANAGER" ]; then
            echo "sdkmanager not found: $SDKMANAGER"
            exit 1
          fi
          "$SDKMANAGER" --install "platforms;android-35" "build-tools;35.0.0"
      - name: Setup Gradle 8.10.2
        uses: gradle/actions/setup-gradle@v4
        with:
          gradle-version: '8.10.2'
      - name: Validate JS
        run: node --check app/src/main/assets/game.js
      - name: Assemble APK
        run: gradle --no-daemon :app:assembleDebug
      - name: Upload APK
        uses: actions/upload-artifact@v4
        with:
          name: SUPER-NIKOLIO-AndroidTV-APK
          path: app/build/outputs/apk/debug/app-debug.apk
          if-no-files-found: error
